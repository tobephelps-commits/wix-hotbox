```typescript
/** Per-angle logo overlay configuration. Each angle is independent. */
export interface AngleOverlayConfig {
  /** Logo name (key in logos.json registry) — shared across all enabled angles */
  logoName: string;
  /** Front angle config (null = no logo on front) */
  front?: {
    position: LogoPosition;
    scale?: number;
    opacity?: number;
    blendMode?: string;
  } | null;
  /** Back angle config (null = no logo on back) */
  back?: {
    position: LogoPosition;
    scale?: number;
    opacity?: number;
    blendMode?: string;
  } | null;
  /** Side angle config (null = no logo on side) */
  side?: {
    position: LogoPosition;
    scale?: number;
    opacity?: number;
    blendMode?: string;
  } | null;
}
```

2. Add new exported function `overlayProductImagesByAngle()` after existing `overlayProductImages()`:

```typescript
/**
 * Apply logo overlay to product images with per-angle configuration.
 * Each angle (front/back/side) gets its own independent placement.
 * Angles set to null or omitted are skipped (no logo applied).
 *
 * @param angleImages - Map of angle to image URL { front?: string, back?: string, side?: string }
 * @param config - Per-angle overlay configuration
 * @param outputDir - Output directory (default: media/overlays/)
 * @returns Map of angle to local overlay file path { front?: string, back?: string, side?: string }
 */
export async function overlayProductImagesByAngle(
  angleImages: { front?: string; back?: string; side?: string },
  config: AngleOverlayConfig,
  outputDir?: string,
): Promise<{ front?: string; back?: string; side?: string }>
```

Implementation: For each angle key (front/back/side), check if `config[angle]` is defined and not null AND `angleImages[angle]` has a URL. If both present, build a `LogoOverlayConfig` from the angle-specific fields + `config.logoName`, call existing `compositeLogoOnImage()`, save to outputDir with `{md5}_overlay_{angle}.jpg` filename. Return map of angle→path for successfully composited images. Skip angles where config is null/undefined or no image URL. Process sequentially like existing `overlayProductImages()`.
