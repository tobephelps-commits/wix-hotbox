```javascript
// =========================================================================
// Bulk Order Selection (Phase 41)
// =========================================================================

var _selectedOrderIds = new Set();

/**
 * Update the bulk action toolbar visibility and selection count.
 */
function updateBulkToolbar() {
  var toolbar = document.getElementById('bulkActionToolbar');
  var countBadge = document.getElementById('bulkSelectionCount');
  var count = _selectedOrderIds.size;

  if (count > 0) {
    toolbar.classList.add('visible');
    countBadge.textContent = count;
  } else {
    toolbar.classList.remove('visible');
  }

  // Update checkbox header state
  var selectAllCb = document.getElementById('selectAllOrders');
  if (selectAllCb) {
    var allRows = document.querySelectorAll('.orders-table tbody .order-checkbox');
    var checkedCount = document.querySelectorAll('.orders-table tbody .order-checkbox:checked').length;
    selectAllCb.indeterminate = checkedCount > 0 && checkedCount < allRows.length;
    selectAllCb.checked = checkedCount > 0 && checkedCount === allRows.length;
  }

  // Enable/disable bulk status button based on dropdown selection
  var statusSelect = document.getElementById('bulkStatusSelect');
  var statusBtn = document.getElementById('bulkStatusBtn');
  statusBtn.disabled = !statusSelect.value || count === 0;
}

/**
 * Toggle selection for a single order.
 */
function toggleOrderSelection(orderId, isSelected) {
  if (isSelected) {
    _selectedOrderIds.add(orderId);
  } else {
    _selectedOrderIds.delete(orderId);
  }

  // Update row visual state
  var row = document.querySelector('.order-row[data-order-id="' + orderId + '"]');
  if (row) {
    row.classList.toggle('selected', isSelected);
  }

  updateBulkToolbar();
}

/**
 * Select or deselect all visible orders.
 */
function toggleAllOrderSelection(selectAll) {
  document.querySelectorAll('.orders-table tbody .order-row').forEach(function(row) {
    var orderId = row.dataset.orderId;
    var cb = row.querySelector('.order-checkbox');
    if (cb) {
      cb.checked = selectAll;
      if (selectAll) {
        _selectedOrderIds.add(orderId);
        row.classList.add('selected');
      } else {
        _selectedOrderIds.delete(orderId);
        row.classList.remove('selected');
      }
    }
  });
  updateBulkToolbar();
}

/**
 * Clear all selections.
 */
function clearOrderSelection() {
  _selectedOrderIds.clear();
  document.querySelectorAll('.orders-table tbody .order-checkbox').forEach(function(cb) {
    cb.checked = false;
  });
  document.querySelectorAll('.orders-table tbody .order-row').forEach(function(row) {
    row.classList.remove('selected');
  });
  var selectAllCb = document.getElementById('selectAllOrders');
  if (selectAllCb) selectAllCb.checked = false;
  updateBulkToolbar();
}

/**
 * Bulk status update action.
 */
async function bulkUpdateStatus() {
  var status = document.getElementById('bulkStatusSelect').value;
  if (!status || _selectedOrderIds.size === 0) return;

  var ids = Array.from(_selectedOrderIds);
  var statusBtn = document.getElementById('bulkStatusBtn');
  statusBtn.disabled = true;
  statusBtn.textContent = 'Updating...';

  try {
    var res = await fetch('/api/orders/bulk/status', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ orderIds: ids, status: status }),
    });
    var json = await res.json();

    var successCount = json.updated ? json.updated.length : 0;
    var failCount = json.failed ? json.failed.length : 0;

    if (successCount > 0) {
      alert('Updated ' + successCount + ' order(s) to "' + status + '"' +
            (failCount > 0 ? '. ' + failCount + ' failed.' : '.'));
      clearOrderSelection();
      loadOrders(); // Refresh the order list
    } else if (failCount > 0) {
      var reasons = json.failed.map(function(f) { return 'Order ' + f.id + ': ' + f.reason; }).join('\n');
      alert('Failed to update orders:\n' + reasons);
    }
  } catch (err) {
    alert('Error updating orders: ' + (err.message || err));
  } finally {
    statusBtn.disabled = false;
    statusBtn.textContent = 'Apply Status';
    document.getElementById('bulkStatusSelect').value = '';
  }
}

/**
 * Bulk production sheets action - download as ZIP.
 */
async function bulkGenerateProductionSheets() {
  if (_selectedOrderIds.size === 0) return;

  var ids = Array.from(_selectedOrderIds);
  var btn = document.getElementById('bulkProductionSheetsBtn');
  btn.disabled = true;
  btn.textContent = 'Generating...';

  try {
    var res = await fetch('/api/orders/bulk/production-sheets/zip', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ orderIds: ids }),
    });

    if (!res.ok) {
      var errJson = await res.json();
      throw new Error(errJson.error || 'Failed to generate production sheets');
    }

    // Download the ZIP file
    var blob = await res.blob();
    var url = window.URL.createObjectURL(blob);
    var a = document.createElement('a');
    a.href = url;
    a.download = 'production-sheets-' + new Date().toISOString().slice(0,10) + '.zip';
    document.body.appendChild(a);
    a.click();
    window.URL.revokeObjectURL(url);
    a.remove();

    alert('Downloaded ' + ids.length + ' production sheet(s)');
  } catch (err) {
    alert('Error generating production sheets: ' + (err.message || err));
  } finally {
    btn.disabled = false;
    btn.textContent = 'Production Sheets';
  }
}

/**
 * Bulk fill SanMar cart - filter selected orders, open cart filler.
 */
function bulkFillSanmarCart() {
  // This reuses existing cart fill logic but pre-filters to selected orders
  var ids = Array.from(_selectedOrderIds);
  if (ids.length === 0) return;

  // Store selection for cart filler to use
  sessionStorage.setItem('bulkCartOrderIds', JSON.stringify(ids));

  // Trigger the existing cart fill button logic with stored filter
  var fillBtn = document.getElementById('fillCartBtn');
  if (fillBtn) {
    fillBtn.click();
  }
}

/**
 * Bulk fill S&S cart - filter selected orders, open cart filler.
 */
function bulkFillSSCart() {
  var ids = Array.from(_selectedOrderIds);
  if (ids.length === 0) return;

  sessionStorage.setItem('bulkCartOrderIds', JSON.stringify(ids));

  var fillBtn = document.getElementById('fillSSCartBtn');
  if (fillBtn) {
    fillBtn.click();
  }
}

// Attach bulk action event listeners on DOMContentLoaded
document.addEventListener('DOMContentLoaded', function() {
  document.getElementById('bulkStatusSelect').addEventListener('change', updateBulkToolbar);
  document.getElementById('bulkStatusBtn').addEventListener('click', bulkUpdateStatus);
  document.getElementById('bulkProductionSheetsBtn').addEventListener('click', bulkGenerateProductionSheets);
  document.getElementById('bulkFillSanmarBtn').addEventListener('click', bulkFillSanmarCart);
  document.getElementById('bulkFillSSBtn').addEventListener('click', bulkFillSSCart);
  document.getElementById('bulkClearBtn').addEventListener('click', clearOrderSelection);
});
```
