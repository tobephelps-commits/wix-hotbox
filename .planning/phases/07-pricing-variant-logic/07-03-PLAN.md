Add pure calculation functions matching the server-side logic:
```javascript
function calcRetailPrice(wholesale, markupPercent, rounding) {
  let price = wholesale * (1 + markupPercent / 100);
  if (rounding === 'nearest-99') price = Math.floor(price) + 0.99;
  else if (rounding === 'nearest-dollar') price = Math.round(price);
  else price = Math.round(price * 100) / 100;
  return price;
}

function calcVariantPrice(wholesale, size, config) {
  return calcRetailPrice(wholesale, config.markupPercent, config.rounding) + (config.sizeUpcharges[size] || 0);
}
```
