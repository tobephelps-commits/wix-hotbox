3. Add JavaScript at the end of the script section (before the closing `</script>` tag):
```javascript
// =========================================================================
// Operations Dashboard (Phase 34)
// =========================================================================

var daemonStartBtn = document.getElementById('daemonStartBtn');
var daemonStopBtn = document.getElementById('daemonStopBtn');
var daemonStatusText = document.getElementById('daemonStatusText');
var opsOrdersNew = document.getElementById('opsOrdersNew');
var opsOrdersTotal = document.getElementById('opsOrdersTotal');
var opsTrackedProducts = document.getElementById('opsTrackedProducts');
var opsRecentAlerts = document.getElementById('opsRecentAlerts');
var opsDaemonCycles = document.getElementById('opsDaemonCycles');
var opsDaemonUptime = document.getElementById('opsDaemonUptime');
var opsErrorCount = document.getElementById('opsErrorCount');
var opsErrorDetail = document.getElementById('opsErrorDetail');

async function loadOperationsHealth() {
  try {
    var res = await fetch('/api/operations/health');
    var data = await res.json();

    // Daemon status and controls
    var running = data.daemon.running;
    daemonStartBtn.disabled = running;
    daemonStopBtn.disabled = !running;
    daemonStatusText.className = 'daemon-status-text ' + (running ? 'running' : 'stopped');
    daemonStatusText.textContent = running ? 'Daemon running' : 'Daemon stopped';

    // Orders card
    var newOrders = data.orders.statusCounts.new || 0;
    opsOrdersNew.textContent = newOrders;
    var totalOrders = Object.values(data.orders.statusCounts).reduce(function(a, b) { return a + b; }, 0);
    opsOrdersTotal.textContent = totalOrders + ' total orders';
    updateCardState('opsOrdersNew', newOrders > 0 ? 'warning' : 'good');

    // Inventory card
    opsTrackedProducts.textContent = data.inventory.trackedProductCount;
    var alertCount = data.inventory.recentAlertCount;
    opsRecentAlerts.textContent = alertCount + ' recent alerts';
    updateCardState('opsTrackedProducts', alertCount > 0 ? 'warning' : 'good');

    // Daemon card
    if (running && data.daemon.health) {
      var health = data.daemon.health;
      opsDaemonCycles.textContent = health.totalCycles;
      opsDaemonUptime.textContent = 'Up ' + formatRelativeTime(new Date(health.startedAt));
      updateCardState('opsDaemonCycles', health.consecutiveErrors > 0 ? 'warning' : 'good');
    } else {
      opsDaemonCycles.textContent = '--';
      opsDaemonUptime.textContent = 'Not running';
      updateCardState('opsDaemonCycles', 'neutral');
    }

    // Errors card
    var errorCount = data.orders.errorCount;
    opsErrorCount.textContent = errorCount;
    opsErrorDetail.textContent = errorCount === 0 ? 'All systems healthy' : 'Order errors detected';
    updateCardState('opsErrorCount', errorCount > 0 ? 'error' : 'good');

  } catch (err) {
    console.error('Failed to load operations health:', err);
    daemonStatusText.textContent = 'Status unknown';
    daemonStatusText.className = 'daemon-status-text';
  }
}

function updateCardState(valueId, state) {
  var card = document.getElementById(valueId).closest('.ops-health-card');
  card.className = 'ops-health-card ' + state;
}

async function startDaemon() {
  daemonStartBtn.disabled = true;
  daemonStatusText.textContent = 'Starting...';
  try {
    var res = await fetch('/api/daemon/start', { method: 'POST' });
    var data = await res.json();
    if (data.success) {
      daemonStatusText.textContent = 'Daemon started';
      daemonStatusText.className = 'daemon-status-text running';
      daemonStopBtn.disabled = false;
      // Refresh health after a moment
      setTimeout(loadOperationsHealth, 1000);
    } else {
      daemonStatusText.textContent = data.message || 'Failed to start';
      daemonStartBtn.disabled = false;
    }
  } catch (err) {
    console.error('Failed to start daemon:', err);
    daemonStatusText.textContent = 'Start failed';
    daemonStartBtn.disabled = false;
  }
}

async function stopDaemon() {
  daemonStopBtn.disabled = true;
  daemonStatusText.textContent = 'Stopping...';
  try {
    var res = await fetch('/api/daemon/stop', { method: 'POST' });
    var data = await res.json();
    if (data.success) {
      daemonStatusText.textContent = 'Daemon stopped';
      daemonStatusText.className = 'daemon-status-text stopped';
      daemonStartBtn.disabled = false;
      // Refresh health after a moment
      setTimeout(loadOperationsHealth, 1000);
    } else {
      daemonStatusText.textContent = data.message || 'Failed to stop';
      daemonStopBtn.disabled = false;
    }
  } catch (err) {
    console.error('Failed to stop daemon:', err);
    daemonStatusText.textContent = 'Stop failed';
    daemonStopBtn.disabled = false;
  }
}

daemonStartBtn.addEventListener('click', startDaemon);
daemonStopBtn.addEventListener('click', stopDaemon);

// Load operations health on page load and refresh periodically
loadOperationsHealth();
setInterval(loadOperationsHealth, 30000); // Refresh every 30 seconds
```
