  <action>
Add module-level daemon control functions for programmatic start/stop:

1. Add module-level state:
```typescript
/** Running sync loop abort controller (null if not running) */
let _abortController: AbortController | null = null;

/** Whether daemon is currently running */
export function isDaemonRunning(): boolean {
  return _abortController !== null && !_abortController.signal.aborted;
}
```

2. Modify startSmartSyncLoop to use AbortController:
- Accept optional `signal?: AbortSignal` parameter
- Store the controller in module state when starting
- Check `signal.aborted` in the tick loop before each iteration
- Clear _abortController and _syncHealth when loop exits

3. Add startDaemon() wrapper function:
```typescript
/**
 * Start the sync daemon programmatically.
 * Returns immediately after starting the loop (non-blocking).
 * Use getSyncHealth() to monitor status, stopDaemon() to stop.
 *
 * @returns true if started, false if already running
 */
export async function startDaemon(): Promise<boolean> {
  if (isDaemonRunning()) return false;

  _abortController = new AbortController();

  // Start loop in background (don't await)
  const monitorConfig = await loadConfig();
  const syncConfig = await getSyncConfigFromEnv(); // extract from manage.ts
  startSmartSyncLoop(monitorConfig, syncConfig, _abortController.signal)
    .catch(err => console.error('[Daemon] Loop error:', err));

  return true;
}
```

4. Add stopDaemon() function:
```typescript
/**
 * Stop the running sync daemon gracefully.
 * Signals the loop to exit after current tick completes.
 *
 * @returns true if stop signal sent, false if not running
 */
export function stopDaemon(): boolean {
  if (!isDaemonRunning()) return false;
  _abortController!.abort();
  return true;
}
```

5. Add getSyncConfigFromEnv() function (extract logic from manage.ts buildSyncConfig):
- Build SyncConfig from environment variables
- Keep in sync-poller.ts since it's now needed for programmatic start

Note: The abort signal approach allows graceful shutdown - the current tick completes before exiting.

3. Add POST /api/daemon/start handler:
```typescript
case 'daemon-start': {
  if (method !== 'POST') {
    sendJson(res, 405, { error: 'Method not allowed. Use POST.' });
    break;
  }
  try {
    const started = await startDaemon();
    if (started) {
      sendJson(res, 200, { success: true, message: 'Daemon started' });
    } else {
      sendJson(res, 409, { success: false, message: 'Daemon already running' });
    }
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    console.error(`[Preview] Error starting daemon: ${message}`);
    sendJson(res, 500, { error: message });
  }
  break;
}
```

4. Add POST /api/daemon/stop handler:
```typescript
case 'daemon-stop': {
  if (method !== 'POST') {
    sendJson(res, 405, { error: 'Method not allowed. Use POST.' });
    break;

    if (stopped) {
```
