async function wizardCreate() {
  const step1 = getWizardStepData(1);
  const step2 = getWizardStepData(2);
  const step3 = getWizardStepData(3);
  const step4 = getWizardStepData(4);

  if (!isWizardReadyToCreate()) {
    alert('Please complete all required steps before creating.');
    return;
  }

  const createBtn = document.getElementById('wizardCreateBtn');
  createBtn.disabled = true;
  createBtn.innerHTML = '&#9203; Creating...';

  try {
    // Build CuratedProduct from wizard state
    const productData = step1.productData;
    const preview = productData.preview;

    // Map selected colors to the format expected by create API
    const selectedColors = step2.selectedColors.map(c => {
      const fullColor = preview.colors.find(pc =>
        pc.displayName === c.displayName || pc.catalogName === c.catalogName
      );
      return {
        displayName: c.displayName,
        catalogName: c.catalogName || fullColor?.catalogName || c.displayName,
        swatchUrl: fullColor?.swatchUrl || '',
        stockStatus: fullColor?.stockStatus || 'unknown'
      };
    });

    const selectedSizes = step2.selectedSizes;

    // Build pricing config
    const pricingConfig = step3.pricingConfig || {
      markupPercent: 100,
      rounding: 'nearest-99',
      sizeUpcharges: {}
    };

    // Build logo config (optional)
    let logoConfig = null;
    if (step4.logoConfig) {
      logoConfig = {
        logoName: step4.logoConfig.logoName,
        front: {
          position: step4.logoConfig.position,
          scale: step4.logoConfig.scale || 0.25
        },
        back: null,
        side: null
      };
    }

    // Get collections
    const collectionsInput = document.getElementById('wizardCollectionsInput');
    const collections = collectionsInput?.value
      ? collectionsInput.value.split(',').map(c => c.trim()).filter(Boolean)
      : [];

    // Build the curated product payload
    const curatedProduct = {
      vendor: step1.vendor,
      style: step1.style,
      selectedColors,
      selectedSizes,
      pricingConfig,
      collections,
      logoConfig  // Will be processed by create endpoint if present
    };

    // Call the create API
    const resp = await fetch('/api/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(curatedProduct)
    });

    if (!resp.ok) {
      const error = await resp.text();
      throw new Error(`Create failed: ${error}`);
    }

    const result = await resp.json();

    // Show success
    showWizardResult(result);

  } catch (err) {
    console.error('Wizard create error:', err);
    alert(`Failed to create product: ${err.message}`);
  } finally {
    createBtn.disabled = false;
    createBtn.innerHTML = '&#10003; Create Product';
  }
}
