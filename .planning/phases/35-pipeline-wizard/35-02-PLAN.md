```javascript
/* Wizard Step 1: Product Selection */
function initWizardStep1() {
  const fetchBtn = document.getElementById('wizardFetchBtn');
  const styleInput = document.getElementById('wizardStyleInput');

  fetchBtn?.addEventListener('click', wizardFetchProduct);
  styleInput?.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') wizardFetchProduct();
  });
}

async function wizardFetchProduct() {
  const vendor = document.getElementById('wizardVendorSelect').value;
  const style = document.getElementById('wizardStyleInput').value.trim().toUpperCase();

  if (!style) return;

  const fetchBtn = document.getElementById('wizardFetchBtn');
  fetchBtn.disabled = true;
  fetchBtn.textContent = 'Fetching...';

  try {
    const resp = await fetch(`/api/product/${encodeURIComponent(style)}?vendor=${vendor}`);
    if (!resp.ok) throw new Error(`Failed to fetch: ${resp.status}`);

    const data = await resp.json();

    // Store in wizard state
    setWizardStepValid(1, true, {
      vendor,
      style,
      productData: data
    });

    // Update preview
    const preview = document.getElementById('wizardProductPreview');
    preview.classList.add('visible');

    document.getElementById('wizardProductName').textContent = data.preview.productName;
    document.getElementById('wizardWholesale').textContent = `$${data.preview.wholesaleCost.toFixed(2)}`;
    document.getElementById('wizardRetail').textContent = `$${data.preview.suggestedRetail.toFixed(2)}`;
    document.getElementById('wizardColorCount').textContent = data.preview.colors.length;
    document.getElementById('wizardSizeCount').textContent = data.preview.sizes.length;

    // Initialize step 2 with product data
    initWizardStep2Content(data.preview);

    // Initialize step 3 preview
    updateWizardPricingPreview();

  } catch (err) {
    console.error('Wizard fetch error:', err);
    alert(`Failed to fetch product: ${err.message}`);
    setWizardStepValid(1, false);
  } finally {
    fetchBtn.disabled = false;
    fetchBtn.textContent = 'Fetch Product';
  }
}

/* Wizard Step 2: Variant Selection */
function initWizardStep2() {
  document.getElementById('wizardSelectAllColors')?.addEventListener('click', () => wizardSelectColors('all'));
  document.getElementById('wizardSelectInStockColors')?.addEventListener('click', () => wizardSelectColors('in-stock'));
  document.getElementById('wizardClearColors')?.addEventListener('click', () => wizardSelectColors('none'));

  document.getElementById('wizardSelectStandardSizes')?.addEventListener('click', () => wizardSelectSizes(['S', 'M', 'L', 'XL']));
  document.getElementById('wizardSelectExtendedSizes')?.addEventListener('click', () => wizardSelectSizes(['S', 'M', 'L', 'XL', '2XL']));
  document.getElementById('wizardSelectAllSizes')?.addEventListener('click', () => wizardSelectSizes('all'));
  document.getElementById('wizardClearSizes')?.addEventListener('click', () => wizardSelectSizes([]));
}

function initWizardStep2Content(preview) {
  // Build color grid
  const colorGrid = document.getElementById('wizardColorGrid');
  colorGrid.innerHTML = preview.colors.map(color => `
    <div class="wizard-color-card" data-color="${color.displayName}" data-catalog="${color.catalogName}">
      <img class="wizard-color-swatch" src="${color.swatchUrl || ''}" alt="${color.displayName}"
           onerror="this.style.background='#30363d'">
      <div class="wizard-color-name">${color.displayName}</div>
      <div class="wizard-color-stock ${color.stockStatus === 'in-stock' ? 'in-stock' : 'out-of-stock'}">
        ${color.stockStatus === 'in-stock' ? '● In Stock' : '○ Low/OOS'}
      </div>
    </div>
  `).join('');

  // Add click handlers
  colorGrid.querySelectorAll('.wizard-color-card').forEach(card => {
    card.addEventListener('click', () => {
      card.classList.toggle('selected');
      updateWizardVariantSelection();
    });
  });

  // Build size grid
  const sizeGrid = document.getElementById('wizardSizeGrid');
  sizeGrid.innerHTML = preview.sizes.map(size => `
    <button type="button" class="wizard-size-btn" data-size="${size}">${size}</button>
  `).join('');

  // Add click handlers
  sizeGrid.querySelectorAll('.wizard-size-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      btn.classList.toggle('selected');
      updateWizardVariantSelection();
    });
  });

  // Reset selections
  updateWizardVariantSelection();
}

function wizardSelectColors(mode) {
  const cards = document.querySelectorAll('.wizard-color-card');
  cards.forEach(card => {
    if (mode === 'all') {
      card.classList.add('selected');
    } else if (mode === 'in-stock') {
      const stock = card.querySelector('.wizard-color-stock');
      card.classList.toggle('selected', stock?.classList.contains('in-stock'));
    } else {
      card.classList.remove('selected');
    }
  });
  updateWizardVariantSelection();
}

function wizardSelectSizes(sizes) {
  const btns = document.querySelectorAll('.wizard-size-btn');
  btns.forEach(btn => {
    const size = btn.dataset.size;
    if (sizes === 'all') {
      btn.classList.add('selected');
    } else if (Array.isArray(sizes)) {
      btn.classList.toggle('selected', sizes.includes(size));
    }
  });
  updateWizardVariantSelection();
}

function updateWizardVariantSelection() {
  const selectedColors = Array.from(document.querySelectorAll('.wizard-color-card.selected'))
    .map(c => ({ displayName: c.dataset.color, catalogName: c.dataset.catalog }));
  const selectedSizes = Array.from(document.querySelectorAll('.wizard-size-btn.selected'))
    .map(b => b.dataset.size);

  const variantCount = selectedColors.length * selectedSizes.length;

  // Update summary
  document.getElementById('wizardSelectionSummary').innerHTML =
    `Selected: <strong>${selectedColors.length} colors</strong> × <strong>${selectedSizes.length} sizes</strong> = <strong>${variantCount} variants</strong>`;

  // Update state
  const isValid = selectedColors.length > 0 && selectedSizes.length > 0;
  setWizardStepValid(2, isValid, { selectedColors, selectedSizes });

  // Update upcharge grid for extended sizes
  updateWizardUpchargeGrid(selectedSizes);
}

/* Wizard Step 3: Pricing */
function initWizardStep3() {
  document.getElementById('wizardPresetSelect')?.addEventListener('change', wizardPresetChanged);
  document.getElementById('wizardMarkupInput')?.addEventListener('input', updateWizardPricingPreview);
  document.getElementById('wizardRoundingSelect')?.addEventListener('change', updateWizardPricingPreview);
}

function wizardPresetChanged() {
  const preset = document.getElementById('wizardPresetSelect').value;
  const markupInput = document.getElementById('wizardMarkupInput');

  const presetMarkups = {
    'standard-tee': 100,
    'premium-tee': 120,
    'hoodie-fleece': 80,
    'polo-woven': 90,
    'outerwear': 70,
    'headwear': 100,
    'custom': parseInt(markupInput.value) || 100
  };

  markupInput.value = presetMarkups[preset] || 100;
  updateWizardPricingPreview();
}

function updateWizardPricingPreview() {
  const step1Data = getWizardStepData(1);
  if (!step1Data.productData) return;

  const wholesale = step1Data.productData.preview.wholesaleCost;
  const markupPercent = parseInt(document.getElementById('wizardMarkupInput').value) || 100;
  const rounding = document.getElementById('wizardRoundingSelect').value;

  // Calculate retail price
  let retail = wholesale * (1 + markupPercent / 100);

  if (rounding === 'nearest-99') {
    retail = Math.ceil(retail) - 0.01;
  } else if (rounding === 'nearest-dollar') {
    retail = Math.ceil(retail);
  }

  const margin = retail - wholesale;
  const marginPercent = ((margin / retail) * 100).toFixed(1);

  document.getElementById('wizardPreviewWholesale').textContent = `$${wholesale.toFixed(2)}`;
  document.getElementById('wizardPreviewMarkup').textContent = `+$${(wholesale * markupPercent / 100).toFixed(2)}`;
  document.getElementById('wizardPreviewRetail').textContent = `$${retail.toFixed(2)}`;
  document.getElementById('wizardPreviewMargin').textContent = `$${margin.toFixed(2)} (${marginPercent}%)`;

  // Store pricing config
  const upcharges = {};
  document.querySelectorAll('.wizard-upcharge-input').forEach(input => {
    const val = parseFloat(input.value) || 0;
    if (val > 0) upcharges[input.dataset.size] = val;
  });

  setWizardStepValid(3, true, {
    pricingConfig: {
      markupPercent,
      rounding,
      sizeUpcharges: upcharges
    }
  });
}

function updateWizardUpchargeGrid(selectedSizes) {
  const extendedSizes = ['2XL', '3XL', '4XL', '5XL', '6XL', 'LT', 'XLT', '2XLT', '3XLT'];
  const relevantSizes = selectedSizes.filter(s => extendedSizes.includes(s));

  const grid = document.getElementById('wizardUpchargeGrid');
  if (relevantSizes.length === 0) {
    grid.innerHTML = '<span style="color:#8b949e;font-size:12px;">No extended sizes selected</span>';
    return;
  }

  grid.innerHTML = relevantSizes.map(size => `
    <div class="wizard-upcharge-item">
      <span>${size}</span>
      <input type="number" class="wizard-upcharge-input" data-size="${size}" value="0" min="0" step="0.50" placeholder="+$">
    </div>
  `).join('');

  // Add listeners
  grid.querySelectorAll('.wizard-upcharge-input').forEach(input => {
    input.addEventListener('input', updateWizardPricingPreview);
  });
}

// Initialize steps on page load
document.addEventListener('DOMContentLoaded', () => {
  initWizardStep1();
  initWizardStep2();
  initWizardStep3();
});
```

```javascript
/* Wizard Step 4: Logo */
function initWizardStep4() {
  const toggle = document.getElementById('wizardLogoEnabled');
  const config = document.getElementById('wizardLogoConfig');

  toggle?.addEventListener('change', () => {
    config.classList.toggle('visible', toggle.checked);
    if (!toggle.checked) {
      // Clear logo selection
      document.querySelectorAll('.wizard-logo-option').forEach(o => o.classList.remove('selected'));
      setWizardStepValid(4, true, { logoConfig: null });
    }
    loadWizardLogos();
  });
}

async function loadWizardLogos() {
  const selector = document.getElementById('wizardLogoSelector');
  if (!selector) return;

  try {
    const resp = await fetch('/api/logos');
    if (!resp.ok) throw new Error('Failed to load logos');

    const logos = await resp.json();
    if (!logos.length) {
      selector.innerHTML = '<div style="color:#8b949e;font-size:13px;">No logos uploaded. Add logos in the Logo Library section.</div>';
      return;
    }

    selector.innerHTML = logos.map(logo => `
      <div class="wizard-logo-option" data-logo="${logo.key}">
        <img src="/data/logos/${logo.filename}" alt="${logo.key}">
        <div class="wizard-logo-option-name">${logo.key}</div>
      </div>
    `).join('');

    selector.querySelectorAll('.wizard-logo-option').forEach(opt => {
      opt.addEventListener('click', () => {
        selector.querySelectorAll('.wizard-logo-option').forEach(o => o.classList.remove('selected'));
        opt.classList.add('selected');
        setWizardStepValid(4, true, {
          logoConfig: {
            logoName: opt.dataset.logo,
            position: { x: 0.5, y: 0.5 },  // Center default
            scale: 0.25
          }
        });
      });
    });
  } catch (err) {
    console.error('Failed to load logos:', err);
    selector.innerHTML = '<div style="color:#f85149;font-size:13px;">Failed to load logos</div>';
  }
}

function updateWizardReview() {
  const step1 = getWizardStepData(1);
  const step2 = getWizardStepData(2);
  const step3 = getWizardStepData(3);
  const step4 = getWizardStepData(4);

  // Product info
  document.getElementById('wizardReviewStyle').textContent = step1.style || '--';
  document.getElementById('wizardReviewVendor').textContent = step1.vendor === 'ss' ? 'S&S Activewear' : 'SanMar';
  document.getElementById('wizardReviewName').textContent = step1.productData?.preview?.productName || '--';

  // Variants
  const colors = step2.selectedColors || [];
  const sizes = step2.selectedSizes || [];
  document.getElementById('wizardReviewColorCount').textContent = colors.length;
  document.getElementById('wizardReviewSizeCount').textContent = sizes.length;
  document.getElementById('wizardReviewVariantCount').textContent = colors.length * sizes.length;

  document.getElementById('wizardReviewColors').innerHTML =
    colors.map(c => `<span class="wizard-review-color-chip">${c.displayName}</span>`).join('');
  document.getElementById('wizardReviewSizes').innerHTML =
    sizes.map(s => `<span class="wizard-review-size-chip">${s}</span>`).join('');

  // Pricing
  const pricing = step3.pricingConfig;
  document.getElementById('wizardReviewMarkup').textContent = pricing ? `${pricing.markupPercent}%` : '--';
  document.getElementById('wizardReviewRounding').textContent = pricing?.rounding || '--';

  if (step1.productData) {
    const wholesale = step1.productData.preview.wholesaleCost;
    let retail = wholesale * (1 + (pricing?.markupPercent || 100) / 100);
    if (pricing?.rounding === 'nearest-99') retail = Math.ceil(retail) - 0.01;
    else if (pricing?.rounding === 'nearest-dollar') retail = Math.ceil(retail);
    document.getElementById('wizardReviewBasePrice').textContent = `$${retail.toFixed(2)}`;
  }

  // Logo
  const logo = step4.logoConfig;
  document.getElementById('wizardReviewLogo').textContent = logo ? `${logo.logoName} (center)` : 'No logo';

  const ready = isWizardReadyToCreate();

  document.getElementById('wizardReviewWarning').style.display = ready ? 'none' : 'block';
}
```
