```javascript
/* ========================================
   Pipeline Wizard (Phase 35)
   ======================================== */

const wizardState = {
  active: false,
  currentStep: 1,
  totalSteps: 5,
  stepData: {
    1: { valid: false, vendor: null, style: null, productData: null },
    2: { valid: false, selectedColors: [], selectedSizes: [] },
    3: { valid: true, pricingConfig: null },  // Defaults always valid
    4: { valid: true, logoConfig: null },      // Optional step, always valid
    5: { valid: false }  // Becomes valid when step 1-4 are valid
  }
};

function initWizard() {
  const startBtn = document.getElementById('wizardStartBtn');
  const closeBtn = document.getElementById('wizardCloseBtn');
  const prevBtn = document.getElementById('wizardPrevBtn');
  const nextBtn = document.getElementById('wizardNextBtn');
  const createBtn = document.getElementById('wizardCreateBtn');
  const stepIndicators = document.querySelectorAll('.wizard-step');

  if (startBtn) startBtn.addEventListener('click', openWizard);
  if (closeBtn) closeBtn.addEventListener('click', closeWizard);
  if (prevBtn) prevBtn.addEventListener('click', wizardPrev);
  if (nextBtn) nextBtn.addEventListener('click', wizardNext);
  if (createBtn) createBtn.addEventListener('click', wizardCreate);

  // Click on completed steps to navigate
  stepIndicators.forEach(step => {
    step.addEventListener('click', () => {
      const stepNum = parseInt(step.dataset.step);
      if (step.classList.contains('completed') || step.classList.contains('active')) {
        goToWizardStep(stepNum);
      }
    });
  });
}

function openWizard() {
  wizardState.active = true;
  wizardState.currentStep = 1;

  // Reset step data
  wizardState.stepData[1] = { valid: false, vendor: null, style: null, productData: null };
  wizardState.stepData[2] = { valid: false, selectedColors: [], selectedSizes: [] };
  wizardState.stepData[3] = { valid: true, pricingConfig: getDefaultPricingConfig() };
  wizardState.stepData[4] = { valid: true, logoConfig: null };
  wizardState.stepData[5] = { valid: false };

  document.getElementById('wizardSection').style.display = 'block';
  document.querySelector('.container').classList.add('wizard-active');

  updateWizardUI();

  // Scroll to wizard
  document.getElementById('wizardSection').scrollIntoView({ behavior: 'smooth', block: 'start' });
}

function closeWizard() {
  wizardState.active = false;
  document.getElementById('wizardSection').style.display = 'none';
  document.querySelector('.container').classList.remove('wizard-active');
}

function getDefaultPricingConfig() {
  return {
    markupPercent: 100,
    rounding: 'nearest-99',
    sizeUpcharges: {}
  };
}

function goToWizardStep(stepNum) {
  if (stepNum < 1 || stepNum > wizardState.totalSteps) return;

  // Can only go forward if current step is valid (or going backward)
  if (stepNum > wizardState.currentStep && !wizardState.stepData[wizardState.currentStep].valid) {
    return;
  }

  wizardState.currentStep = stepNum;
  updateWizardUI();
}

function wizardPrev() {
  if (wizardState.currentStep > 1) {
    goToWizardStep(wizardState.currentStep - 1);
  }
}

function wizardNext() {
  if (wizardState.currentStep < wizardState.totalSteps) {
    if (wizardState.stepData[wizardState.currentStep].valid) {
      goToWizardStep(wizardState.currentStep + 1);
    }
  }
}

async function wizardCreate() {
  // Implemented in Plan 35-03
  console.log('Wizard create - to be implemented');
}

function updateWizardUI() {
  const currentStep = wizardState.currentStep;

  // Update step indicators
  document.querySelectorAll('.wizard-step').forEach(el => {
    const stepNum = parseInt(el.dataset.step);
    el.classList.remove('active', 'completed');

    if (stepNum === currentStep) {
      el.classList.add('active');
    } else if (stepNum < currentStep || wizardState.stepData[stepNum].valid) {
      el.classList.add('completed');
    }
  });

  // Update panels
  document.querySelectorAll('.wizard-panel').forEach(el => {
    const stepNum = parseInt(el.dataset.step);
    el.classList.toggle('active', stepNum === currentStep);
  });

  // Update navigation buttons
  const prevBtn = document.getElementById('wizardPrevBtn');
  const nextBtn = document.getElementById('wizardNextBtn');
  const createBtn = document.getElementById('wizardCreateBtn');

  prevBtn.disabled = currentStep === 1;

  // Show Create button only on last step
  if (currentStep === wizardState.totalSteps) {
    nextBtn.style.display = 'none';
    createBtn.style.display = 'flex';
    createBtn.disabled = !isWizardReadyToCreate();
  } else {
    nextBtn.style.display = 'flex';
    createBtn.style.display = 'none';
    nextBtn.disabled = !wizardState.stepData[currentStep].valid;
  }

  // Update step 5 validity based on steps 1-4
  wizardState.stepData[5].valid =
    wizardState.stepData[1].valid &&
    wizardState.stepData[2].valid;
}

function isWizardReadyToCreate() {
  return wizardState.stepData[1].valid &&
         wizardState.stepData[2].valid;
  // Steps 3 and 4 always valid (have defaults)
}

function setWizardStepValid(stepNum, valid, data = {}) {
  wizardState.stepData[stepNum].valid = valid;
  Object.assign(wizardState.stepData[stepNum], data);
  updateWizardUI();
}

function getWizardStepData(stepNum) {
  return wizardState.stepData[stepNum];
}

// Initialize wizard on page load
document.addEventListener('DOMContentLoaded', () => {
  initWizard();
});
```

This provides:
1. `wizardState` - tracks current step, validation state, and collected data per step
2. `openWizard()` / `closeWizard()` - toggle wizard mode
3. `goToWizardStep(n)` - navigate to specific step (only if allowed)
4. `wizardPrev()` / `wizardNext()` - navigation with validation gates
5. `updateWizardUI()` - sync UI with state (indicators, panels, buttons)
6. `setWizardStepValid()` / `getWizardStepData()` - API for step panels to update state

Do NOT modify any existing functions - this is additive only.
